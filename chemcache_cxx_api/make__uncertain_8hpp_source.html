37inline simde::type::tensor make_uncertain(const simde::type::tensor& t,
38 const simde::UQFactory& factory) {
39 if(factory.kind() == simde::UQKind::none) return t;
40
41 using tensorwrapper::buffer::get_raw_data;
42 using tensorwrapper::buffer::make_contiguous;
43 const auto& buffer = make_contiguous(t.buffer());
44 const auto values = get_raw_data<double>(buffer);
45 auto shape = t.buffer().layout().shape().as_smooth().make_smooth();
46
47 return simde::visit_uq_kind(factory.kind(), [&](auto type) {
48 using float_type = typename decltype(type)::type;
49 std::vector<float_type> data;
50 data.reserve(values.size());
51 for(const auto x : values) {
52 auto value = factory(x, 0.0);
53 data.push_back(wtf::fp::float_cast<float_type>(value));
54 }
55 tensorwrapper::buffer::Contiguous uq_buffer(std::move(data), shape);
56 return simde::type::tensor(shape, std::move(uq_buffer));
57 });
58}
